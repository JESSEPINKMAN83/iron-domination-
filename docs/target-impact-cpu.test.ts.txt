import { it, vi } from 'vitest';
import { PerspectiveCamera, Scene } from 'three';
import { UnitView } from './unitView';
import { BuildingView } from './buildingView';
import { UnitView as BaseUnits } from '../../../target-impact-baseline/src/render/unitView';
import { BuildingView as BaseBuildings } from '../../../target-impact-baseline/src/render/buildingView';
import { createGameSim, spawnTankAt } from '../sim/world';
import { createEconomy, placeStructure } from '../sim/economy';
import { applyStructureDamage } from '../sim/structureDamage';
import { generateHeightfield } from '../sim/heightfield';
import { MAP01 } from '../content/map01';
import type { RenderContext } from './renderer';
it('measures matched target damage CPU work', () => {
 const context=new Proxy({}, {get:(_,key)=>key==='createRadialGradient'||key==='createLinearGradient'?()=>({addColorStop:()=>undefined}):key==='measureText'?()=>({width:100}):key==='createImageData'?(w:number,h:number)=>({data:new Uint8ClampedArray(w*h*4)}):key==='getImageData'?(_x:number,_y:number,w:number,h:number)=>({data:new Uint8ClampedArray(w*h*4)}):()=>undefined});
 vi.stubGlobal('document',{createElement:()=>({width:0,height:0,getContext:()=>context})});
 const hf=generateHeightfield(MAP01);hf.heights.fill(0);hf.walkable.fill(1);hf.waterLevel=-10;
 const ctx={setupLitMaterial:<T>(m:T):T=>m} as unknown as RenderContext;
 const camera=new PerspectiveCamera(45,16/9,.1,1000);camera.position.set(90,65,115);camera.lookAt(0,0,0);camera.updateMatrixWorld();
 const results:Record<string,number[]>={baseSteady:[],headSteady:[],baseCollapse:[],headCollapse:[]};
 for(let trial=0;trial<10;trial++) for(const [name,U,B] of (trial%2?[['head',UnitView,BuildingView],['base',BaseUnits,BaseBuildings]]:[['base',BaseUnits,BaseBuildings],['head',UnitView,BuildingView]]) as Array<[string,typeof UnitView,typeof BuildingView]>) {
  const sim=createGameSim(hf),eco=createEconomy(2,1000000),scene=new Scene();
  const units=Array.from({length:24},(_,i)=>{const e=spawnTankAt(sim,-40+i%8*10,20+Math.floor(i/8)*10,'Target',2);e.health!.current=e.health!.max*.24;return e;});
  const buildings=Array.from({length:8},(_,i)=>{eco.readyStructure='factory';const e=placeStructure(sim,hf,eco,{kind:'factory',x:-60+i%4*38,z:-50+Math.floor(i/4)*36,valid:true,reason:''})!;e.building!.buildProgress=1;e.health!.current=e.health!.max*.4;applyStructureDamage(e,{hitX:e.transform.x,hitZ:e.transform.z,fromX:e.transform.x+50,fromZ:e.transform.z,amount:300,splashRadius:6});return e;});
  const u=new U(units,hf,ctx),b=new B(sim,hf,ctx);u.setCombatChromeVisible(false);b.setCombatChromeVisible(false);scene.add(u.group,b.group);
  u.update(1,1/60,camera);b.update(eco,camera);scene.updateMatrixWorld();
  let start=performance.now();
  for(let frame=0;frame<120;frame++){sim.tick=Math.floor(frame/2);u.update(1,1/60,camera);b.update(eco,camera);scene.updateMatrixWorld();}
  const steady=performance.now()-start;
  for(const e of buildings){e.health!.current=0;e.destroyed={remaining:20};}
  start=performance.now();
  for(let frame=0;frame<90;frame++){sim.tick=60+Math.floor(frame/2);for(const e of buildings)e.destroyed!.remaining=20-Math.floor(frame/2)/30;u.update(1,1/60,camera);b.update(eco,camera);scene.updateMatrixWorld();}
  const collapse=performance.now()-start;
  if(trial>2){results[name+'Steady'].push(steady);results[name+'Collapse'].push(collapse);}
 }
 console.log('TARGET_CPU',Object.fromEntries(Object.entries(results).map(([key,data])=>[key,{median:data.sort((a,b)=>a-b)[Math.floor(data.length/2)],samples:data}])));
 vi.unstubAllGlobals();
},60000);
