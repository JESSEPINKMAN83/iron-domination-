import * as before from "/tmp/iron-move-base-nlwoopq3/src/sim/world";

import {generateHeightfield} from "/Users/danir/Documents/Codex/2026-11-10/previous-chats-where-is-the-iron/work/iron-dominion/src/sim/heightfield";
function run(api) {
 const hf=generateHeightfield({cells:96,cellSize:2,waterLevel:-100,seed:41,oreFieldCount:0});hf.heights.fill(0);hf.walkable.fill(1);
 const sim=api.createGameSim(hf);const units=Array.from({length:36},(_,i)=>api.spawnTankAt(sim,-75+i%6*8,-75+Math.floor(i/6)*8,'Formation',1));
 api.issueMoveOrder(sim,units,35,35);const start=performance.now();
 for(let i=0;i<2400;i++)api.stepSim(sim,hf,1/30);
 return {ms:performance.now()-start,arrived:units.filter(e=>!e.mover.target).length};
}

for(let i=0;i<10;i++)for(const key of (i%2?['after','before']:['before','after'])){const r=run(key==='before'?before:after);if(i>=3)samples[key].push(r);}
console.log(JSON.stringify(samples));
